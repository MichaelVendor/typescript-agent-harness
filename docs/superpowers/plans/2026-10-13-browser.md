# 内置浏览器（Playwright MCP）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 修好 MCP stdio 客户端，并内置 Playwright MCP：平时只挂 `browser_open`，调用后才启动 Chrome、挂上 18 个浏览器工具。

**Architecture:** `packages/mcp` 改为按行收发 JSON 并把 `tools/call` 结果转成文字；`packages/tools` 的 ToolService 加 `unregister`；`packages/cli/src/browser.ts` 提供 `createBrowser()`（插件 + `isOpen()`），`bootRuntime` 按 `capabilities.browser` 挂上，审批的 `ask` 在浏览器已开时对 `browser_open` 直接放行。

**Tech Stack:** TypeScript (Node ≥22, ES2022)、`node:test`、`@playwright/mcp@0.0.83`、React（packages/web）。

## Global Constraints

- 设计：`docs/design/browser.md`。
- `@playwright/mcp` 锁定 `0.0.83`，作为 `packages/cli` 依赖；启动用 `node <包目录>/<bin["playwright-mcp"]>`（`cli.js` 不在 exports 里，先 resolve `package.json`）。

- [ ] **Step 1: 改 fixture 为按行收发**，每条消息分两次 `write`（覆盖分块拼接），新增 `fail`（`isError: true`）和 `shot`（text + image）两个工具：

```js
#!/usr/bin/env node
/** Minimal MCP stdio server for tests: one JSON-RPC message per line, as the MCP spec says. */
let buffer = "";
function write(msg) {
  const line = `${JSON.stringify(msg)}\n`;
  const half = Math.floor(line.length / 2);
  process.stdout.write(line.slice(0, half));
  setImmediate(() => process.stdout.write(line.slice(half)));
}
const TOOLS = [
  { name: "ping", description: "echo ping", inputSchema: { type: "object", properties: { text: { type: "string" } } } },
  { name: "fail", description: "always fails", inputSchema: { type: "object", properties: {} } },
  { name: "shot", description: "text and an image", inputSchema: { type: "object", properties: {} } },
];
function handle(msg) {
  if (msg.method === "initialize") {
    return write({ jsonrpc: "2.0", id: msg.id, result: { protocolVersion: "2024-11-05", capabilities: { tools: {} }, serverInfo: { name: "ah-test-ping", version: "0.0.0" } } });
  }
  if (msg.method === "tools/list") return write({ jsonrpc: "2.0", id: msg.id, result: { tools: TOOLS } });
  if (msg.method !== "tools/call") return;
  const name = msg.params?.name;
  if (name === "ping") {
    return write({ jsonrpc: "2.0", id: msg.id, result: { content: [{ type: "text", text: `pong:${msg.params?.arguments?.text ?? ""}` }] } });
  }
  if (name === "fail") return write({ jsonrpc: "2.0", id: msg.id, result: { content: [{ type: "text", text: "boom" }], isError: true } });
  if (name === "shot") {
    return write({ jsonrpc: "2.0", id: msg.id, result: { content: [{ type: "text", text: "saved" }, { type: "image", data: "AAAA", mimeType: "image/png" }] } });
  }
  write({ jsonrpc: "2.0", id: msg.id, error: { code: -32601, message: `unknown tool: ${name}` } });
}
process.stdin.setEncoding("utf8");
process.stdin.on("data", (chunk) => {
  buffer += chunk;
  let nl;
  while ((nl = buffer.indexOf("\n")) !== -1) {
    const line = buffer.slice(0, nl).trim();
    buffer = buffer.slice(nl + 1);
    if (line) handle(JSON.parse(line));
  }
});
process.stdin.on("end", () => process.exit(0));
```

- [ ] **Step 4: 实现 `stdio-backend.ts`**：`write` 改为 `child.stdin.write(JSON.stringify(msg) + "\n")`；`stdout.setEncoding("utf8")`，按 `\n` 切行、空行跳过、`JSON.parse` 失败忽略；带 `method` 的消息（服务器发来的请求 / 通知）忽略；`stderr` 只保留最后 2000 字符，进程退出时拼进错误；`child.on("error")` 和 `child.stdin.on("error")` 让等待中的请求全部 reject。

```ts
  let stdoutBuf = "";
  let stderrTail = "";

  function write(msg: JsonRpc) {
    if (!child) throw new Error("MCP stdio process is not running");
    child.stdin.write(`${JSON.stringify(msg)}\n`);
  }

  function failAll(err: Error) {
    for (const waiter of pending.values()) waiter.reject(err);
    pending.clear();
  }

  function onMessage(msg: JsonRpc) {
    if (msg.method !== undefined || msg.id === undefined) return;
    const waiter = pending.get(msg.id);
    if (!waiter) return;
    pending.delete(msg.id);
    if (msg.error) waiter.reject(new Error(msg.error.message));
    else waiter.resolve(msg.result);
  }

  function ingest(chunk: string) {
    stdoutBuf += chunk;
    let nl: number;
    while ((nl = stdoutBuf.indexOf("\n")) !== -1) {
      const line = stdoutBuf.slice(0, nl).trim();
      stdoutBuf = stdoutBuf.slice(nl + 1);
      if (!line) continue;
      let msg: JsonRpc;
      try {
        msg = JSON.parse(line) as JsonRpc;
      } catch {
        continue;
      }
      onMessage(msg);
    }
  }
```

```ts
/** MCP `tools/call` result → text for the model: text parts joined, other parts noted, `isError` thrown. */
export function mcpResultText(result: unknown): unknown {
  const content = (result as { content?: unknown } | null)?.content;
  if (!Array.isArray(content)) return result;
  const text = (content as Array<{ type?: string; text?: string }>)
    .map((p) => (p.type === "text" ? (p.text ?? "") : `[${p.type ?? "unknown"} omitted]`))
    .join("\n");
  if ((result as { isError?: boolean }).isError) throw new Error(text || "MCP tool failed");
  return text;
}
```

```ts
import { readFileSync } from "node:fs";
import { createRequire } from "node:module";
import path from "node:path";
import type { Plugin } from "@typescript-agent-harness/core";
import { mcpResultText, stdioMcpBackend, type McpBackend } from "@typescript-agent-harness/mcp";
import { TOOLS, type Tool, type ToolService } from "@typescript-agent-harness/tools";

/** Playwright MCP tools the model gets; code-running, network-inspection and rarely useful ones stay off. */
export const BROWSER_TOOLS = [
  "browser_navigate", "browser_navigate_back", "browser_snapshot", "browser_find",
  "browser_click", "browser_type", "browser_fill_form", "browser_select_option",
  "browser_press_key", "browser_hover", "browser_drag", "browser_wait_for",
  "browser_tabs", "browser_take_screenshot", "browser_handle_dialog", "browser_file_upload",
  "browser_console_messages", "browser_close",
];

export const BROWSER_RESULT_CHARS = 30_000;

export type BrowserServer = { command: string; args: string[] };

export type BrowserOptions = {
  cwd: string;
  homeDir: string;
  /** `tah run` has nobody watching; chat and serve show the window. */
  headless: boolean;
  /** Tests swap in a fake MCP server. */
  server?: BrowserServer;
};

export function playwrightServer(o: BrowserOptions): BrowserServer {
  const require = createRequire(import.meta.url);
  const pkg = require.resolve("@playwright/mcp/package.json");
  const bin = (JSON.parse(readFileSync(pkg, "utf8")) as { bin: Record<string, string> }).bin["playwright-mcp"]!;
  return {
    command: process.execPath,
    args: [
      path.join(path.dirname(pkg), bin),
      "--browser", "chrome",
      "--user-data-dir", path.join(o.homeDir, ".tah", "browser-profile"),
      "--output-dir", path.join(o.cwd, ".tah", "browser"),
      "--image-responses", "omit",
      ...(o.headless ? ["--headless"] : []),
    ],
  };
}

export function clipResult(out: unknown): unknown {
  if (typeof out !== "string" || out.length <= BROWSER_RESULT_CHARS) return out;
  return `${out.slice(0, BROWSER_RESULT_CHARS)}\n[truncated: ${out.length - BROWSER_RESULT_CHARS} more chars — use browser_find to search the page]`;
}

/** `browser_open` starts Playwright MCP on first use and mounts the browser tools; `browser_close` takes them down again. */
export function createBrowser(o: BrowserOptions): { plugin: Plugin; isOpen(): boolean } {
  let tools: ToolService | undefined;
  let backend: McpBackend | undefined;
  let opening: Promise<McpBackend> | undefined;
  let mounted: string[] = [];

  async function shut() {
    for (const name of mounted) tools?.unregister(name);
    mounted = [];
    const b = backend;
    backend = undefined;
    await b?.close?.();
  }

  async function start(): Promise<McpBackend> {
    const b = stdioMcpBackend({ ...(o.server ?? playwrightServer(o)), cwd: o.cwd });
    try {
      const infos = (await b.listTools()).filter((t) => BROWSER_TOOLS.includes(t.name));
      for (const info of infos) {
        tools!.register({
          name: info.name,
          description: info.description,
          inputSchema: info.inputSchema,
          async execute(input) {
            const out = clipResult(mcpResultText(await b.callTool(info.name, input)));
            if (info.name === "browser_close") await shut();
            return out;
          },
        });
        mounted.push(info.name);
      }
      backend = b;
      return b;
    } catch (err) {
      for (const name of mounted) tools!.unregister(name);
      mounted = [];
      await b.close?.();
      throw new Error(`could not start the browser: ${err instanceof Error ? err.message : String(err)}`);
    }
  }

  const openTool: Tool = {
    name: "browser_open",
    description:
      "Open a Chrome window the user can see (or reuse the open one) and go to `url` (a blank page if omitted). Afterwards the browser_* tools (navigate, snapshot, click, type, fill_form, …) are available. Use it for clicking, forms, logins, or pages rendered by JavaScript; use web_fetch to just read a page.",
    inputSchema: {
      type: "object",
      properties: { url: { type: "string", description: "Page to open; omit for a blank page" } },
      additionalProperties: false,
    },
    async execute(input) {
      const wasOpen = backend !== undefined;
      const b = backend ?? (await (opening ??= start().finally(() => (opening = undefined))));
      const url = (input as { url?: unknown } | undefined)?.url;
      const target = typeof url === "string" && url ? url : wasOpen ? undefined : "about:blank";
      const page = target ? mcpResultText(await b.callTool("browser_navigate", { url: target })) : "";
      return { opened: !wasOpen, tools: [...mounted], page: clipResult(page) };
    },
  };

  return {
    plugin: {
      name: "browser",
```
